# MS3 Slices 3A + 3B Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement interactive editor onboarding tour (3A) and visual AI diff in draft modal (3B) in parallel.

**Architecture:** 3A adds a custom spotlight overlay component using DOM rect positioning with `data-tour-id` anchors. 3B wires existing `WorkflowDiffView`/`diffWorkflowGraphs` components into `AiDraftModal` as a new "Visual Diff" tab with per-node accept/reject. Both are client-side only — no backend changes.

**Tech Stack:** React 18, TypeScript, Zustand 5, XYFlow 12 (React Flow), @phosphor-icons/react, vitest + @testing-library/react

## Global Constraints

- No new npm dependencies for 3A (custom implementation using DOM APIs)
- No new npm dependencies for 3B (reuse existing diff components)
- `data-tour-id` attributes as stable anchors (NOT CSS class names)
- localStorage key: `noodle-editor-tour-v1` for dismiss persistence
- Tour must work at 1280px and 1920px viewport widths
- All new components must have corresponding vitest tests

```typescript
import { useState, useCallback } from "react";

export interface TourStep {
  id: string;
  target: string; // CSS selector for data-tour-id
  title: string;
  body: string;
  position: "center" | "right" | "bottom" | "left" | "top";
}

export const TOUR_STEPS: readonly TourStep[] = [
  {
    id: "canvas",
    target: '[data-tour-id="canvas"]',
    title: "Your workflow canvas",
    body: "Drag nodes here to build automation workflows. Double-click anywhere to quick-add a node.",
    position: "center",
  },
  {
    id: "palette",
    target: '[data-tour-id="palette"]',
    title: "Node palette",
    body: "Browse 70+ built-in nodes — HTTP, code, AI, data, integrations. Drag any node onto the canvas.",
    position: "right",
  },
  {
    id: "toolbar",
    target: '[data-tour-id="toolbar"]',
    title: "Quick-add shortcut",
    body: "Press Tab anywhere on the canvas to open the node search and insert a node at your cursor.",
    position: "bottom",
  },
  {
    id: "run",
    target: '[data-tour-id="run-button"]',
    title: "Run your workflow",
    body: "Click Run to execute all nodes. Live output appears on each node tile as it completes.",
    position: "bottom",
  },
  {
    id: "ai",
    target: '[data-tour-id="ai-draft-button"]',
    title: "AI workflow builder",
    body: "Type what you want to automate in plain English. AI generates the full workflow graph — editable.",
    position: "bottom",
  },
] as const;

const STORAGE_KEY = "noodle-editor-tour-v1";

export function useOnboardingTour() {
  const [currentStep, setCurrentStep] = useState(0);
  const [isActive, setIsActive] = useState(() => {
    return localStorage.getItem(STORAGE_KEY) !== "true";
  });

  const totalSteps = TOUR_STEPS.length;
  const currentStepDef = TOUR_STEPS[currentStep] ?? null;
  const isFirstStep = currentStep === 0;
  const isLastStep = currentStep === totalSteps - 1;

  const next = useCallback(() => {
    setCurrentStep((prev) => {
      const nextIdx = prev + 1;
      if (nextIdx >= totalSteps) {
        setIsActive(false);
        localStorage.setItem(STORAGE_KEY, "true");
        return prev;
      }
      return nextIdx;
    });
  }, [totalSteps]);

  const prev = useCallback(() => {
    setCurrentStep((p) => Math.max(0, p - 1));
  }, []);

  const skip = useCallback(() => {
    setIsActive(false);
    localStorage.setItem(STORAGE_KEY, "true");
  }, []);

  const restart = useCallback(() => {
    setCurrentStep(0);
    setIsActive(true);
    localStorage.removeItem(STORAGE_KEY);
  }, []);

  return {
    isActive,
    currentStep,
    totalSteps,
    currentStepDef,
    next,
    prev,
    skip,
    restart,
    isFirstStep,
    isLastStep,
  } as const;
}
```

```typescript
import { useState, useEffect, useCallback, useRef } from "react";
import { useOnboardingTour } from "./useOnboardingTour";
import type { TourStep } from "./useOnboardingTour";

interface TargetRect {
  top: number;
  left: number;
  width: number;
  height: number;
}

const TOOLTIP_OFFSET = 16;
const SPOTLIGHT_PADDING = 8;

function getTooltipStyle(
  rect: TargetRect,
  position: TourStep["position"],
): React.CSSProperties {
  const base: React.CSSProperties = { position: "fixed", zIndex: 10001 };
  const gap = TOOLTIP_OFFSET;

  switch (position) {
    case "center":
      return {
        ...base,
        top: "50%",
        left: "50%",
        transform: "translate(-50%, -50%)",
      };
    case "bottom":
      return {
        ...base,
        top: rect.top + rect.height + gap,
        left: rect.left + rect.width / 2,
        transform: "translateX(-50%)",
      };
    case "top":
      return {
        ...base,
        bottom: window.innerHeight - rect.top + gap,
        left: rect.left + rect.width / 2,
        transform: "translateX(-50%)",
      };
    case "right":
      return {
        ...base,
        top: rect.top + rect.height / 2,
        left: rect.left + rect.width + gap,
        transform: "translateY(-50%)",
      };
    case "left":
      return {
        ...base,
        top: rect.top + rect.height / 2,
        left: rect.left - gap,
        transform: "translate(-100%, -50%)",
      };
  }
}

function getSpotlightStyle(rect: TargetRect): React.CSSProperties {
  const p = SPOTLIGHT_PADDING;
  if (rect.width === 0 && rect.height === 0) {
    // Center position — spotlight the center area
    return {
      position: "fixed",
      top: "calc(50% - 120px)",
      left: "calc(50% - 200px)",
      width: "400px",
      height: "240px",
      borderRadius: "12px",
      boxShadow: "0 0 0 9999px rgba(0, 0, 0, 0.5)",
      zIndex: 10000,
      pointerEvents: "none",
    };
  }
  return {
    position: "fixed",
    top: rect.top - p,
    left: rect.left - p,
    width: rect.width + p * 2,
    height: rect.height + p * 2,
    borderRadius: "8px",
    boxShadow: "0 0 0 9999px rgba(0, 0, 0, 0.5)",
    zIndex: 10000,
    pointerEvents: "none",
  };
}

export function OnboardingTour() {
  const { isActive, currentStep, totalSteps, currentStepDef, next, prev, skip, isFirstStep, isLastStep } = useOnboardingTour();
  const [targetRect, setTargetRect] = useState<TargetRect | null>(null);
  const [targetNotFound, setTargetNotFound] = useState(false);
  const pollRef = useRef<number>(0);
  const pollCountRef = useRef(0);

  const findTarget = useCallback(() => {
    if (!currentStepDef) return;
    const el = document.querySelector(currentStepDef.target) as HTMLElement | null;
    if (el) {
      const r = el.getBoundingClientRect();
      // Check if element has non-zero dimensions and is in viewport
      if (r.width > 0 || r.height > 0 || currentStepDef.position === "center") {
        setTargetRect({ top: r.top, left: r.left, width: r.width, height: r.height });
        setTargetNotFound(false);
        return;
      }
      // Element exists but is zero-size — scroll into view and retry
      el.scrollIntoView({ behavior: "instant", block: "center" });
    }
    // Poll with requestAnimationFrame (max 2s)
    pollCountRef.current += 1;
    if (pollCountRef.current < 120) {
      pollRef.current = requestAnimationFrame(findTarget);
    } else {
      setTargetNotFound(true);
    }
  }, [currentStepDef]);

  useEffect(() => {
    if (!isActive || !currentStepDef) return;
    pollCountRef.current = 0;
    setTargetRect(null);
    setTargetNotFound(false);
    // Small delay to let any layout shifts settle
    const timer = setTimeout(findTarget, 100);
    return () => {
      clearTimeout(timer);
      cancelAnimationFrame(pollRef.current);
    };
  }, [isActive, currentStep, currentStepDef, findTarget]);

  // ResizeObserver for viewport changes
  useEffect(() => {
    if (!isActive) return;
    const onResize = () => {
      pollCountRef.current = 0;
      findTarget();
    };
    window.addEventListener("resize", onResize);
    return () => window.removeEventListener("resize", onResize);
  }, [isActive, findTarget]);

  if (!isActive || !currentStepDef) return null;

  return (
    <>
      {/* Spotlight cut-out */}
      {targetRect && !targetNotFound && (
        <div style={getSpotlightStyle(targetRect)} />
      )}
      {/* Tooltip */}
      {!targetNotFound && (
        <div
          style={targetRect ? getTooltipStyle(targetRect, currentStepDef.position) : { position: "fixed", top: "50%", left: "50%", transform: "translate(-50%, -50%)", zIndex: 10001 }}
          className="onboarding-tooltip"
        >
          <div className="onboarding-tooltip-step">
            Step {currentStep + 1} of {totalSteps}
          </div>
          <h3 className="onboarding-tooltip-title">{currentStepDef.title}</h3>
          <p className="onboarding-tooltip-body">{currentStepDef.body}</p>
          <div className="onboarding-tooltip-actions">
            <button type="button" className="btn" onClick={skip}>
              Skip tour
            </button>
            <div className="onboarding-tooltip-nav">
              {!isFirstStep && (
                <button type="button" className="btn" onClick={prev}>
                  Previous
                </button>
              )}
              <button type="button" className="btn is-primary" onClick={next}>
                {isLastStep ? "Finish" : "Next"}
              </button>
            </div>
          </div>
        </div>
      )}
      {/* Target not found fallback */}
      {targetNotFound && (
        <div
          style={{ position: "fixed", top: "50%", left: "50%", transform: "translate(-50%, -50%)", zIndex: 10001 }}
          className="onboarding-tooltip"
        >
          <div className="onboarding-tooltip-step">
            Step {currentStep + 1} of {totalSteps}
          </div>
          <h3 className="onboarding-tooltip-title">{currentStepDef.title}</h3>
          <p className="onboarding-tooltip-body">{currentStepDef.body}</p>
          <p className="onboarding-tooltip-note">
            (The highlighted element is not currently visible. It will appear when you interact with the editor.)
          </p>
          <div className="onboarding-tooltip-actions">
            <button type="button" className="btn" onClick={skip}>
              Skip tour
            </button>
            <div className="onboarding-tooltip-nav">
              {!isFirstStep && (
                <button type="button" className="btn" onClick={prev}>
                  Previous
                </button>
              )}
              <button type="button" className="btn is-primary" onClick={next}>
                {isLastStep ? "Finish" : "Next"}
              </button>
            </div>
          </div>
        </div>
      )}
    </>
  );
}
```

```typescript
// Add to the existing Props interface — a second props shape:
interface GraphDiffViewProps {
  baseGraph: WorkflowGraph;
  compareGraph: WorkflowGraph;
  onClose?: () => void;
  // Per-node accept/reject callbacks
  rejectedNodeIds?: Set<string>;
  onToggleReject?: (nodeId: string) => void;
  readOnly?: boolean;
}

// Extract the internal diff canvas that works on plain graphs
export function GraphDiffView({
  baseGraph,
  compareGraph,
  onClose,
  rejectedNodeIds,
  onToggleReject,
  readOnly,
}: GraphDiffViewProps) {
  const diff = useMemo(() => diffWorkflowGraphs(baseGraph, compareGraph), [baseGraph, compareGraph]);

  const statusMap = useMemo(() => {
    const map = new Map<string, DiffStatus>();
    for (const id of diff.added) map.set(id, "added");
    for (const id of diff.removed) map.set(id, "removed");
    for (const id of diff.changed) map.set(id, "changed");
    for (const id of diff.unchanged) map.set(id, "unchanged");
    return map;
  }, [diff]);

  const [selectedNodeId, setSelectedNodeId] = useState<string | null>(null);

  // Build render nodes/edges from compareGraph, applying status colors
  const { renderNodes, renderEdges } = useMemo(() => {
    const nodes = compareGraph.nodes
      .filter((n) => !rejectedNodeIds?.has(n.id) || diff.added.includes(n.id) === false)
      .map((n) => {
        const status = statusMap.get(n.id) ?? "unchanged";
        const isRejected = rejectedNodeIds?.has(n.id);
        return {
          id: n.id,
          type: "diffNode",
          position: (n as any).position ?? { x: 0, y: 0 },
          data: {
            ...n,
            isRejected,
            onToggleReject: onToggleReject ? () => onToggleReject(n.id) : undefined,
            readOnly,
          },
        };
      });

    const validIds = new Set(nodes.map((n) => n.id));
    const edges = compareGraph.edges
      .filter((e) => validIds.has(e.source) && validIds.has(e.target))
      .map((e) => {
        const isAdded = diff.addedEdges.includes(e.id);
        const isRemoved = diff.removedEdges.includes(e.id);
        return {
          ...e,
          type: isAdded ? "diffEdgeAdded" : isRemoved ? "diffEdgeRemoved" : "diffEdge",
        };
      });

    return { renderNodes: nodes, renderEdges: edges };
  }, [compareGraph, diff, statusMap, rejectedNodeIds, onToggleReject, readOnly]);

  const selectedParams = selectedNodeId ? (diff.changedParams[selectedNodeId] ?? null) : null;

  return (
    <DiffContext.Provider value={statusMap}>
      <div style={{ width: "100%", height: "100%", position: "relative" }}>
        <ReactFlow
          nodes={renderNodes}
          edges={renderEdges}
          nodeTypes={diffNodeTypes}
          edgeTypes={diffEdgeTypes}
          fitView
          nodesDraggable={false}
          nodesConnectable={false}
          elementsSelectable={true}
          onNodeClick={(_event, node) => setSelectedNodeId(node.id)}
        >
          <Background />
          <Panel position="top-left">
            <DiffSummaryBar
              added={diff.added.length}
              removed={diff.removed.length}
              changed={diff.changed.length}
              unchanged={diff.unchanged.length}
            />
          </Panel>
        </ReactFlow>
        {selectedParams && selectedParams.length > 0 && selectedNodeId && (
          <NodeParamDiffPanel
            nodeId={selectedNodeId}
            params={selectedParams}
            onClose={() => setSelectedNodeId(null)}
          />
        )}
        {onClose && (
          <button
            type="button"
            className="diff-close-btn"
            onClick={onClose}
            style={{ position: "absolute", top: 12, right: 12, zIndex: 10 }}
            aria-label="Close diff view"
          >
            ✕
          </button>
        )}
      </div>
    </DiffContext.Provider>
  );
}
```

```typescript
function applyWithFilters(
  currentGraph: WorkflowGraph,
  proposedGraph: WorkflowGraph,
  rejectedNodeIds: Set<string>,
): WorkflowGraph {
  const currentById = new Map(currentGraph.nodes.map((n) => [n.id, n]));
  const proposedById = new Map(proposedGraph.nodes.map((n) => [n.id, n]));

  const resultNodes: GraphNode[] = [];
  const resultNodeIds = new Set<string>();

  for (const node of proposedGraph.nodes) {
    const isRejected = rejectedNodeIds.has(node.id);
    if (!currentById.has(node.id)) {
      // Added node
      if (!isRejected) {
        resultNodes.push(node);
        resultNodeIds.add(node.id);
      }
    } else if (isRejected) {
      // Modified node rejected — restore original
      resultNodes.push(currentById.get(node.id)!);
      resultNodeIds.add(node.id);
    } else {
      // Accepted (new or unchanged)
      resultNodes.push(node);
      resultNodeIds.add(node.id);
    }
  }

  // Restore removed nodes that were rejected
  for (const node of currentGraph.nodes) {
    if (!proposedById.has(node.id) && rejectedNodeIds.has(node.id)) {
      resultNodes.push(node);
      resultNodeIds.add(node.id);
    }
  }

  // Filter edges
  const filteredEdges = proposedGraph.edges
    .filter((e) => resultNodeIds.has(e.source) && resultNodeIds.has(e.target));
  const restoredEdges = currentGraph.edges
    .filter(
      (e) =>
        resultNodeIds.has(e.source) &&
        resultNodeIds.has(e.target) &&
        !filteredEdges.some((fe) => fe.id === e.id),
    );

  return {
    nodes: resultNodes,
    edges: [...filteredEdges, ...restoredEdges],
  };
}
```
